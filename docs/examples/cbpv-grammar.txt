value  ->  Var sym
        |  Int i
        |  I64 i
        |  U64 i
        |  Float f
        |  Bool b
        |  Unit
        |  Str s
        |  Thunk comp
        |  Ctor sym tag [ value, ... ]
        |  Tuple [ value, ... ]
        |  UnboxedTuple [ value, ... ]
        |  UnboxedRecord [ (sym, value), ... ]

comp   ->  Return value
        |  Bind comp sym comp
        |  Force value
        |  Lam [ sym, ... ] comp
        |  App comp [ value, ... ]
        |  If value comp comp
        |  Prim coreop value value
        |  Call sym [ value, ... ]
        |  Io ioop [ value, ... ]
        |  Error value
        |  Case value [ (pat, comp), ... ]
        |  FloatBuiltin floatop value
        |  Neg neglane value
        |  UnboxedProject value sym
        |  Do sym [ value, ... ]
        |  Handle { body: comp, return_var: maybe-sym,
                    return_body: maybe-comp, ops: [ handleop, ... ] }
        |  Mask [ sym, ... ] comp
        |  StrBuiltin builtin [ value, ... ]
        |  Dup value
        |  Drop value
        |  WithReuse { token: sym, freed: value, body: comp }
        |  Reuse sym value
        |  InitAt value value
        |  RefNew value
        |  RefGet value
        |  RefSet value value

pat    ->  Wild
        |  Var sym
        |  Ctor sym [ maybe-sym, ... ]
        |  Tuple [ maybe-sym, ... ]

maybe-sym -> Some sym | None
maybe-comp -> Some comp | None

handleop ->  HandleOp { name: sym, params: [ sym, ... ],
                        resume: sym, body: comp }

corefn ->  CoreFn { name: sym, params: [ sym, ... ], body: comp,
                    dict_arity: n }

core   ->  Core [ corefn, ... ]
